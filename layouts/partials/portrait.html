{{- /* dict: params (person .Params), size (px), crop (bool, default true) */ -}} {{- $p := .params -}} {{- $size := int .size -}} {{- $size2 := mul $size 2 -}} {{- $crop := true -}} {{- if isset . "crop" -}}{{- $crop = .crop -}}{{- end -}} {{- $src := resources.Get "img/flower.png" -}} {{- $isPhoto := false -}} {{- with $p.detail -}} {{- with .photo -}} {{- with resources.Get . -}}{{- $src = . -}}{{- $isPhoto = true -}}{{- end -}} {{- end -}} {{- end -}} {{- $alt := "" -}} {{- if and $isPhoto $p.personName -}}{{- $alt = $p.personName -}}{{- end -}} {{- /* The flower placeholder and uncropped photos keep their aspect ratio (fit / resize-by-width); cropped photos are squared off (fill). Each is emitted as resized WebP 1x/2x with a same-size JPEG/PNG fallback via ; width/height come from the processed resource. Hugo hashes every variant's filename, so they cache safely. */ -}} {{- $ext := cond (eq $src.MediaType.SubType "png") "png" "jpeg" -}} {{- $sizing := "" -}} {{- $sizing2 := "" -}} {{- if not $isPhoto -}} {{- $sizing = printf "fit %dx%d" $size $size -}} {{- $sizing2 = printf "fit %dx%d" $size2 $size2 -}} {{- else if $crop -}} {{- $sizing = printf "fill %dx%d" $size $size -}} {{- $sizing2 = printf "fill %dx%d" $size2 $size2 -}} {{- else -}} {{- $sizing = printf "resize %dx" $size -}} {{- $sizing2 = printf "resize %dx" $size2 -}} {{- end -}} {{- $w1 := $src.Process (printf "%s %s q82" $sizing $ext) -}} {{- $w2 := $src.Process (printf "%s %s q82" $sizing2 $ext) -}} {{- $p1 := $src.Process (printf "%s webp q82" $sizing) -}} {{- $p2 := $src.Process (printf "%s webp q82" $sizing2) -}} {{- $class := "flower" -}} {{- if $isPhoto -}}{{- $class = cond $crop "flower photo" "flower photo-full" -}}{{- end -}}
